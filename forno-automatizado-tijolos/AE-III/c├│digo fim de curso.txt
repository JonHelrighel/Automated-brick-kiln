#include "hal/adc_types.h"
#include "hal/gpio_types.h"
#include "soc/gpio_num.h"
#include <stdint.h>
#include <stdio.h>
#include <stdbool.h>
#include <sys/dirent.h>
#include <unistd.h>
#include <driver/gpio.h>
#include <esp_adc/adc_oneshot.h>

#define MOTOR_E GPIO_NUM_19
#define MOTOR_D GPIO_NUM_18
#define COMANDO GPIO_NUM_5

void app_main(void)
{
	gpio_set_direction(MOTOR_D, GPIO_MODE_OUTPUT);
	gpio_set_direction(MOTOR_E, GPIO_MODE_OUTPUT);
	gpio_set_direction(COMANDO, GPIO_MODE_INPUT);
	
	adc_oneshot_unit_handle_t adc2_handle;
	adc_oneshot_unit_init_cfg_t init_config1 = {
	    .unit_id = ADC_UNIT_2,
	    .ulp_mode = ADC_ULP_MODE_DISABLE,
	};
	
	adc_oneshot_new_unit(&init_config1, &adc2_handle);
	
	adc_oneshot_chan_cfg_t config = {
	    .bitwidth = ADC_BITWIDTH_DEFAULT,
	    .atten = ADC_ATTEN_DB_0,
	};
	
	adc_oneshot_config_channel(adc2_handle, 9, &config);
	
	int adc_read = 10;
	int med = 0;
	int reads = 6;
	uint8_t input = 0;

	uint8_t aux = 0;
	uint8_t move_porta = 0;
	uint8_t direcao = 0;
	uint8_t aux_fim_curso = 0;
	
    while (true) {
		
		for(int i = 0; i < reads; i++){
			adc_oneshot_read(adc2_handle, 9, &adc_read);
			med += adc_read;
		}
		
		med = med/reads;
		
		if(med > 1900){
			input = 0b00000011;
		
		}else if (med > 1500){
			input = 0b00000010;
		
		}else if (med > 600){
			input = 0b00000001;
		
		}else{
			input = 0b00000000;
		}
		
		//Inicia o controle da porta
		if(gpio_get_level(COMANDO) == 1 && aux == 0){
			move_porta = !move_porta;
		}
		aux = gpio_get_level(COMANDO);
		
		//liga o motor da porta
		if(move_porta == 1){
			if(direcao){
				gpio_set_level(MOTOR_D, 1);
				gpio_set_level(MOTOR_E, 0);	
			}else {
				gpio_set_level(MOTOR_D, 0);
				gpio_set_level(MOTOR_E, 1);
			}
		}else {
			gpio_set_level(MOTOR_D, 0);
			gpio_set_level(MOTOR_E, 0);
		}
		
		//inverte direção com valor do fim de curso e desliga motor na mudaça de fim de curso
		switch (input) {
			case 1:
				direcao = 1;
				if(aux_fim_curso == 2){
					move_porta = 0;
				}
				aux_fim_curso = 1;
			break;
			
			case 2:
				direcao = 0;
				if(aux_fim_curso == 1){
					move_porta = 0;
				}
				aux_fim_curso = 2;
			break;
		}
		
        printf("valor ADC: %d\n", med);
		printf("input: %d\n", input);
		printf("abre: %d direção: %d aux_liga: %d aux_porta: %d\n", move_porta, direcao, aux, aux_fim_curso);
        usleep(100000);
    }
}
